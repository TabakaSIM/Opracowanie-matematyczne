	\section{Potwórzenie wiadomości z działań na potęgach}
	
	\ZbiorZadanieTwoXThree{Oblicz:}{$2^3$}{$5^2$}{$2^8$}{$\frac{1}{2}^3$}{$\sqrt{2}^4$}{$1^{24}$}
	
	Dla $x,y \in \mathbb{Z}$ oraz $a,b\neq0$ zachodzą następujące wzory:
	
	\Wzor{$a^x\cdot a^y=a^{x+y} \qquad a^x: a^y=\frac{a^x}{a^y}=a^{x-y}$
	$$(a\cdot b)^x=a^x \cdot b^x \qquad \left(\frac{a}{b}\right)^x=(a:b)^x=a^x:b^x=\frac{a^x}{b^x}$$
	$$\left( a^x\right) ^y = a^{x\cdot y}$$}
